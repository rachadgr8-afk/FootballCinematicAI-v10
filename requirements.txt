# CPU-safe Python dependencies for FootballCinematicAI ULTRA
# The CPU PyTorch index is an extra index so normal PyPI packages remain available.
--extra-index-url https://download.pytorch.org/whl/cpu

# Official PyTorch CPU wheel pair
torch==2.6.0+cpu
torchvision==0.21.0+cpu

ultralytics>=8.3.0

# Pinned so the OpenCV wheel and numpy stay ABI-compatible: opencv-python 4.11+
# wheels are built against numpy>=2, while torch 2.6+cpu resolves numpy<2. A
# mismatch makes `import cv2` fail at runtime, which would silently force the
# local analysis into fallback mode.
#
# This pair ALSO satisfies the standalone `video_engine.py` (class CinematicEngine),
# which only needs OpenCV + NumPy. Do NOT add a second/competing OpenCV wheel
# (opencv-contrib-python*): mixing two OpenCV builds in the same venv breaks
# `import cv2`. The fourcc `mp4v` writer and Farneback optical flow used by the
# engine are part of the standard (non-contrib) build already pinned here.
numpy>=1.26,<2
opencv-python-headless==4.10.0.84

# RIFE is optional and NOT a Python package: it is invoked as a subprocess from a
# separately cloned repository (see RIFE_INSTALL.md / RIFE_REPO env var). Installing
# it via pip fails because the upstream repo has no setup.py/pyproject.toml.

pytest>=8.3,<9

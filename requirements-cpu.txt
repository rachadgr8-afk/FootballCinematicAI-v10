# CPU-safe Python dependencies for FootballCinematicAI ULTRA
# The CPU PyTorch index is an extra index so normal PyPI packages remain available.
--extra-index-url https://download.pytorch.org/whl/cpu

# Official PyTorch CPU wheel pair
torch==2.6.0+cpu
torchvision==0.21.0+cpu

ultralytics>=8.3.0

# Pinned so the OpenCV wheel and numpy stay ABI-compatible: opencv-python 4.11+
# wheels are built against numpy>=2, while torch 2.6+cpu resolves numpy<2. A
# mismatch makes `import cv2` fail at runtime, which would silently force the
# local analysis into fallback mode.
numpy>=1.26,<2
opencv-python-headless==4.10.0.84
